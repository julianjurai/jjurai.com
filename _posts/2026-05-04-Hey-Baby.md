<script>
// Global Game State
let currentGame = null;
let score = 0;
let soundEnabled = true;
let difficulty = 'easy'; // easy, medium, hard

// Speech Synthesis
function speak(text) {
  if (!soundEnabled || !window.speechSynthesis) return;

  const utterance = new SpeechSynthesisUtterance(text);
  utterance.rate = 0.8;
  utterance.pitch = 1.2;
  utterance.volume = 1;
  speechSynthesis.speak(utterance);
}

// Settings
function toggleSound() {
  soundEnabled = !soundEnabled;
  document.getElementById('soundButton').textContent = soundEnabled ? '🔊 Sound On' : '🔇 Sound Off';
  if (soundEnabled) speak('Sound is on');
}

function changeDifficulty() {
  const levels = ['easy', 'medium', 'hard'];
  const currentIndex = levels.indexOf(difficulty);
  difficulty = levels[(currentIndex + 1) % levels.length];
  document.getElementById('difficultyButton').textContent = `Level: ${difficulty.charAt(0).toUpperCase() + difficulty.slice(1)}`;
  speak(`Level ${difficulty}`);
}

// Navigation
function startGame(gameType) {
  currentGame = gameType;
  score = 0;
  document.getElementById('menuScreen').style.display = 'none';
  document.getElementById('gameScreen').classList.add('active');

  const titles = {
    feedMonster: '🦖 Feed the Monster',
    animalFinder: '🐘 Animal Finder',
    numberGarden: '🔢 Number Garden',
    shapeSorter: '🔷 Shape Sorter',
    colorCatch: '🎨 Color Catch'
  };

  document.getElementById('gameTitle').textContent = titles[gameType];
  updateScore();

  // Start the appropriate game
  switch(gameType) {
    case 'feedMonster':
      playFeedMonster();
      break;
    case 'animalFinder':
      playAnimalFinder();
      break;
    case 'numberGarden':
      playNumberGarden();
      break;
    case 'shapeSorter':
      playShapeSorter();
      break;
    case 'colorCatch':
      playColorCatch();
      break;
  }
}

function backToMenu() {
  document.getElementById('gameScreen').classList.remove('active');
  document.getElementById('menuScreen').style.display = 'block';
  currentGame = null;
  score = 0;
}

function updateScore() {
  document.getElementById('scoreDisplay').textContent = `Score: ${score}`;
}

function showFeedback(message, isCorrect) {
  const feedbackArea = document.getElementById('feedbackArea');
  feedbackArea.textContent = message;
  feedbackArea.style.color = isCorrect ? '#48c774' : '#f14668';
  speak(message);

  if (isCorrect) {
    showCelebration();
  }
}

function showCelebration() {
  const celebration = document.createElement('div');
  celebration.className = 'celebration';
  celebration.textContent = ['🎉', '⭐', '✨', '🌟', '💫'][Math.floor(Math.random() * 5)];
  document.body.appendChild(celebration);
  setTimeout(() => celebration.remove(), 1000);
}

// Game 1: Feed the Monster
function playFeedMonster() {
  const gameArea = document.getElementById('gameArea');
  const promptArea = document.getElementById('promptArea');
  const feedbackArea = document.getElementById('feedbackArea');
  feedbackArea.textContent = '';

  const foods = {
    easy: [
      { emoji: '🍎', name: 'apple', color: 'red' },
      { emoji: '🍌', name: 'banana', color: 'yellow' },
      { emoji: '🍊', name: 'orange', color: 'orange' },
      { emoji: '🍇', name: 'grapes', color: 'purple' }
    ],
    medium: [
      { emoji: '🍎', name: 'apple', color: 'red', count: 1 },
      { emoji: '🍌', name: 'banana', color: 'yellow', count: 2 },
      { emoji: '🍊', name: 'orange', color: 'orange', count: 3 },
      { emoji: '🍇', name: 'grapes', color: 'purple', count: 1 }
    ],
    hard: [
      { emoji: '🍎', name: 'apple', color: 'red', count: 2 },
      { emoji: '🍌', name: 'banana', color: 'yellow', count: 3 },
      { emoji: '🍊', name: 'orange', color: 'orange', count: 1 },
      { emoji: '🍇', name: 'grapes', color: 'purple', count: 2 }
    ]
  };

  const currentFoods = foods[difficulty];
  let correctFood;

  if (difficulty === 'easy') {
    correctFood = currentFoods[Math.floor(Math.random() * currentFoods.length)];
    promptArea.textContent = `Feed me the ${correctFood.name}!`;
    speak(`Feed me the ${correctFood.name}`);
  } else {
    correctFood = currentFoods[Math.floor(Math.random() * currentFoods.length)];
    promptArea.textContent = `Feed me ${correctFood.count} ${correctFood.name}${correctFood.count > 1 ? 's' : ''}!`;
    speak(`Feed me ${correctFood.count} ${correctFood.name}${correctFood.count > 1 ? 's' : ''}`);
  }

  gameArea.innerHTML = `
    <div class="monster" id="monster">🦖</div>
    <div class="food-tray" id="foodTray"></div>
  `;

  const foodTray = document.getElementById('foodTray');
  const shuffled = [...currentFoods].sort(() => Math.random() - 0.5);

  shuffled.forEach(food => {
    const foodItem = document.createElement('div');
    foodItem.className = 'food-item';
    foodItem.textContent = food.emoji;
    foodItem.onclick = () => checkFoodChoice(food, correctFood);
    foodTray.appendChild(foodItem);
  });
}

function checkFoodChoice(chosen, correct) {
  const monster = document.getElementById('monster');

  if (difficulty === 'easy') {
    if (chosen.name === correct.name) {
      monster.classList.add('eating');
      setTimeout(() => monster.classList.remove('eating'), 500);
      score++;
      updateScore();
      showFeedback('Yummy! Thank you!', true);
      setTimeout(() => playFeedMonster(), 2000);
    } else {
      showFeedback(`Oops! I asked for a ${correct.name}.`, false);
    }
  } else {
    // For medium/hard, simplified to just check the food type
    if (chosen.name === correct.name) {
      monster.classList.add('eating');
      setTimeout(() => monster.classList.remove('eating'), 500);
      score++;
      updateScore();
      showFeedback('Yummy! Thank you!', true);
      setTimeout(() => playFeedMonster(), 2000);
    } else {
      showFeedback(`Oops! I asked for ${correct.name}.`, false);
    }
  }
}

// Game 3: Number Garden
function playNumberGarden() {
  const gameArea = document.getElementById('gameArea');
  const promptArea = document.getElementById('promptArea');
  const feedbackArea = document.getElementById('feedbackArea');
  feedbackArea.textContent = '';

  const maxNum = difficulty === 'easy' ? 3 : difficulty === 'medium' ? 5 : 10;
  const correctNum = Math.floor(Math.random() * maxNum) + 1;

  const objects = ['🍎', '⭐', '🐟', '🎈', '🚗', '🦆', '🍪'];
  const chosenObject = objects[Math.floor(Math.random() * objects.length)];

  let objectDisplay = '';
  for (let i = 0; i < correctNum; i++) {
    objectDisplay += chosenObject + ' ';
  }

  promptArea.textContent = 'How many do you see?';
  speak('How many do you see?');

  const numOptions = difficulty === 'easy' ? 3 : difficulty === 'medium' ? 4 : 5;
  const numbers = [];
  numbers.push(correctNum);
  while (numbers.length < numOptions) {
    const rand = Math.floor(Math.random() * maxNum) + 1;
    if (!numbers.includes(rand)) numbers.push(rand);
  }
  numbers.sort(() => Math.random() - 0.5);

  gameArea.innerHTML = `
    <div style="font-size: 3em; margin-bottom: 30px;">${objectDisplay}</div>
    <div class="options-area" id="optionsArea"></div>
  `;

  const optionsArea = document.getElementById('optionsArea');
  numbers.forEach(num => {
    const button = document.createElement('button');
    button.className = 'option-button';
    button.textContent = num;
    button.onclick = () => checkNumberChoice(num, correctNum);
    optionsArea.appendChild(button);
  });
}

function checkNumberChoice(chosen, correct) {
  if (chosen === correct) {
    score++;
    updateScore();
    showFeedback(`Yes! ${correct}!`, true);
    setTimeout(() => playNumberGarden(), 2000);
  } else {
    showFeedback(`Let's count together. The answer is ${correct}.`, false);
  }
}
